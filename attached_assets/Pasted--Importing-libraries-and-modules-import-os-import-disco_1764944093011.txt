# --- Helper Functions for Data Management ---


def load_banned_list():
    """Loads the list of banned keywords from a JSON file."""
    try:
        with open(BANNED_LIST_FILE, 'r') as f:
            data = json.load(f)
            # Convert loaded lists of keywords to sets for efficient lookup
            return {
                guild_id: set(keywords)
                for guild_id, keywords in data.items()
            }
    except (FileNotFoundError, json.JSONDecodeError):
        # Return an empty dictionary if file not found or corrupted
        return {}


def save_banned_list(banned_data):
    """Saves the current banned keywords to a JSON file."""
    with open(BANNED_LIST_FILE, 'w') as f:
        # Convert sets back to lists for JSON serialization
        data_to_save = {
            guild_id: list(keywords)
            for guild_id, keywords in banned_data.items()
        }
        json.dump(data_to_save, f, indent=4)


def load_playlists():
    """Loads user-created playlists from a JSON file."""
    try:
        with open(PLAYLISTS_FILE, 'r') as f:
            return json.load(f)
    except (FileNotFoundError, json.JSONDecodeError):
        # Return an empty dictionary if file not found or corrupted
        return {}


def save_playlists(playlists_data):
    """Saves the current user playlists to a JSON file."""
    with open(PLAYLISTS_FILE, 'w') as f:
        json.dump(playlists_data, f, indent=4)

# --- Autocomplete Functions ---
async def playlist_autocompleter(
    interaction: discord.Interaction,
    current: str,
) -> List[app_commands.Choice[str]]:
    """Reusable autocompleter for playlist names."""
    guild_id = str(interaction.guild.id)
    playlists = PLAYLISTS.get(guild_id, {})
    all_playlist_names = list(playlists.keys())

    if not current:
        # If user hasn't typed anything, show all playlists (up to 25)
        return [
            app_commands.Choice(name=name, value=name)
            for name in all_playlist_names[:25]
        ]
    else:
        # Filter based on user input (case-insensitive)
        filtered_playlists = [
            name for name in all_playlist_names
            if current.lower() in name.lower()
        ]
        return [
            app_commands.Choice(name=name, value=name)
            for name in filtered_playlists[:25]
        ]


async def song_title_autocompleter(

    current: str,
) -> List[app_commands.Choice[str]]:

            app_commands.Choice(name=title, value=title)

        ]
    else:
        # Filter based on user input (case-insensitive)
        filtered_song_titles = [
            title for title in all_song_titles
            if current.lower() in title.lower()
        ]
        return [
            app_commands.Choice(name=title, value=title)
            for title in filtered_song_titles[:25]
        ]

# --- Keyword Checking Logic ---


def is_banned(guild_id: str, query: str) -> bool:
    """
    Checks if the given query contains any banned keywords for the guild.
    Normalizes both query and banned keywords by removing non-alphanumeric characters and spaces.
    Returns True if a banned keyword is found, False otherwise.
    """
    # Compile a regex pattern to match any non-alphanumeric character, underscore, or space.
    # re.UNICODE flag ensures correct handling of international characters (like Chinese).
    pattern = re.compile(r'[\W_ ]', re.UNICODE)

    # Normalize the user's query: remove non-alphanumeric characters/spaces and convert to lowercase
    normalized_query = pattern.sub('', query).lower()

    if guild_id in BANNED_KEYWORDS:
        for banned_word in BANNED_KEYWORDS[guild_id]:
            # Normalize each banned keyword using the same pattern
            normalized_banned_word = pattern.sub('', banned_word).lower()
            # Check if any normalized banned word is a substring of the normalized query
            if normalized_banned_word in normalized_query:
                return True  # Indicate that a banned keyword was found
    return False  # Indicate no banned keyword was found


async def check_banned_keywords(interaction: discord.Interaction, query: str):
    """
    Checks if the given query contains any banned keywords for the guild.
    Normalizes both query and banned keywords by removing non-alphanumeric characters and spaces.
    """
    guild_id_str = str(interaction.guild.id)
    if is_banned(guild_id_str, query):
        await interaction.response.send_message(
            f"誰准你查這個!", ephemeral=True)  # Send a private warning
        return True  # Indicate that a banned keyword was found
    return False  # Indicate no banned keyword was found


# --- YouTube-DLP Related Functions ---


async def search_ytdlp_async(query, ydl_opts):
    """Executes a yt-dlp search asynchronously in a separate thread."""
    loop = asyncio.get_running_loop()
    return await loop.run_in_executor(None, lambda: _extract(query, ydl_opts))


def _extract(query, ydl_opts):
    """Helper function to perform the actual yt-dlp extraction synchronously."""
    with yt_dlp.YoutubeDL(ydl_opts) as ydl:
        return ydl.extract_info(query, download=False)

# Load Opus library for voice functionality
import subprocess
import os

def load_opus():
    """Load the Opus audio codec library required for Discord voice functionality."""
    if discord.opus.is_loaded():
        print("✅ Opus already loaded!")
        return True
        
    print("🔍 Loading Opus library...")
    
    # Method 1: Try to use pkg-config to find libopus
    try:
        result = subprocess.run(['pkg-config', '--libs', 'opus'], 
                              capture_output=True, text=True, timeout=5)
        if result.returncode == 0:
            # Extract library path from pkg-config output
            lib_path = result.stdout.strip().replace('-lopus', 'libopus.so')
            if os.path.exists(lib_path):
                try:
                    discord.opus.load_opus(lib_path)
                    print(f"✅ Loaded Opus via pkg-config: {lib_path}")
                    return True
                except OSError:
                    pass
    except (subprocess.TimeoutExpired, FileNotFoundError):
        pass
    
    # Method 2: Try common library names (let system find the path)
    common_names = [
        'libopus.so.0',  # Most common name
        'libopus.so',    # Alternative name
        'opus',          # Simple name
    ]
    
    for name in common_names:
        try:
            discord.opus.load_opus(name)
            print(f"✅ Loaded Opus from system: {name}")
            return True
        except OSError:
            continue
    
    # Method 3: Try some specific paths as fallback
    fallback_paths = [
        '/usr/lib/x86_64-linux-gnu/libopus.so.0',
        '/usr/lib/libopus.so.0',
        '/usr/lib/libopus.so',
    ]
    
    for path in fallback_paths:
        try:
            if os.path.exists(path):
                discord.opus.load_opus(path)
                print(f"✅ Loaded Opus from path: {path}")
                return True
        except OSError:
            continue
    
    print("❌ Failed to load Opus library - voice functionality may not work")
    print("ℹ️  The bot will still work for text commands, but audio may have issues")
    return False

# Load Opus before creating the bot

async def search_song(query: str):
    """Searches for a song on YouTube using yt-dlp and returns its URL, title, and headers."""
    ydl_options = {
        "format":
        "bestaudio[abr<=96]/bestaudio",  # Prioritize efficient audio format
        "noplaylist": True,  # Prevent downloading entire playlists
        "youtube_include_dash_manifest":
        False,  # Optimize for faster info extraction
        "youtube_include_hls_manifest": False,
        "extractor_args": {
            "youtube":
            ["player_client=web"]  # Use web client for broader compatibility
        },
        "quiet": True,  # Suppress yt-dlp output
        'cookiefile': 'cookies.txt',
    }
    search_query = "ytsearch1: " + query  # Prefix query to search on YouTube
    try:
        results = await search_ytdlp_async(search_query, ydl_options)
        tracks = results.get("entries", [])
        if not tracks:
            return None, "找不到這首歌耶！"  # Return error if no results found
        first_track = tracks[0]
        audio_url = first_track.get("url")
        title = first_track.get("title", "Untitled")
        
        # 【!! 403 修復 1 !!】 取得 yt-dlp 提供的 http_headers
        headers = first_track.get("http_headers")
        
        # 【!! 403 修復 2 !!】 在回傳值中加入 headers
        return (audio_url, title, headers), None  # Return audio URL, title, and headers

    except Exception as e:
        print(f"搜尋時發生錯誤：{e}")  # Log any errors during search
        return None, f"搜尋時發生錯誤：{e}"

@bot.tree.command(name="play", description="輸入關鍵字")
@app_commands.describe(song_query="搜尋關鍵字")
async def play(interaction: discord.Interaction, song_query: str):
    """Plays a song from a search query."""
    if not interaction.user.voice or not interaction.user.voice.channel:
        return await send_or_replace(interaction, "你先進語音嘛！")

    # 【10062 修復】 - Defer response first
    await interaction.response.defer()

    # 【10062 修復】 - Check for banned keywords
    if is_banned(str(interaction.guild.id), song_query):
        await interaction.followup.send(f"誰准你查這個!", ephemeral=True)
        return  # Exit if a banned keyword is found
    voice_channel = interaction.user.voice.channel
    voice_client = interaction.guild.voice_client
    if voice_client is None:
        voice_client = await voice_channel.connect(
        )  # Connect to voice channel
    elif voice_channel != voice_client.channel:
        await voice_client.move_to(voice_channel
                                   )  # Move to user's voice channel

    song_info, error_message = await search_song(song_query)
    if error_message:
        await interaction.followup.send(error_message)
        return

    audio_url, title, headers = song_info
    guild_id = str(interaction.guild_id)
    if SONG_QUEUES.get(guild_id) is None:
        SONG_QUEUES[guild_id] = deque()  # Initialize queue if not exists

    SONG_QUEUES[guild_id].append((audio_url, title, headers))  # Add song to queue with headers

    if voice_client.is_playing() or voice_client.is_paused():
        await interaction.followup.send(f"加入待播清單: **{title}**")
    else:
        await interaction.followup.send(f"正在播放: **{title}**")
        await play_next_song(voice_client, guild_id, interaction.channel)


async def play_next_song(voice_client, guild_id, channel):
    """Plays the next song in the queue for a given guild."""
    if guild_id in SONG_QUEUES and SONG_QUEUES[guild_id]:
        
        # 【!! 403 修復 5 !!】 從佇列中解開 headers
        audio_url, title, headers = SONG_QUEUES[guild_id].popleft()
        
        # 調試輸出
        print(f"🎵 準備播放: {title}")
        print(f"📦 Headers 類型: {type(headers)}")
        if headers:
            print(f"📦 Headers 內容: {list(headers.keys())[:5] if isinstance(headers, dict) else 'Not a dict'}")
        
        # 建立基本的 FFmpeg 選項
        before_options = "-reconnect 1 -reconnect_streamed 1 -reconnect_delay_max 5"
        
        # 【!! 403 修復 6 !!】 如果有 headers，將它們序列化到 before_options
        if headers and isinstance(headers, dict):
            # 將 headers 字典轉換為 FFmpeg 需要的格式：-headers "Key1: Value1\r\nKey2: Value2\r\n"
            headers_str = "".join([f"{key}: {value}\r\n" for key, value in headers.items()])
            before_options += f' -headers "{headers_str}"'
            print(f"✅ 已添加 headers 到 FFmpeg")
        else:
            print(f"⚠️ 沒有 headers 或格式錯誤")
        
        ffmpeg_options = {
            "before_options": before_options,
            "options": "-vn -filter:a 'volume=0.5'"  # Remove video, add volume filter to reduce volume and improve quality
        }
        source = discord.FFmpegPCMAudio(audio_url, **ffmpeg_options)

        def after_play(error):
            """Callback function to play next song after current one finishes or errors."""
            if error:
                print(f"播放錯誤 {title}: {error}")
            bot.loop.create_task(
                play_next_song(voice_client, guild_id, channel))

        voice_client.play(source, after=after_play)  # Start playing the song
        key = (channel.guild.id, channel.id)
        if key in LAST_MESSAGES:
            try:
                await LAST_MESSAGES[key].delete(
                )  # Delete previous "Now Playing" message
            except discord.NotFound:
                pass  # Message already deleted or not found
        msg = await channel.send(f"正在播放: **{title}**"
                                 )  # Send new "Now Playing" message
        LAST_MESSAGES[key] = msg
    else:
        # If queue is empty, disconnect from voice channel
        if voice_client.is_connected():
            await voice_client.disconnect()
        if guild_id in SONG_QUEUES:
            SONG_QUEUES[guild_id].clear()  # Ensure queue is clear

@bot.tree.command(name="add", description="將歌曲加入播放清單")

@bot.tree.command(name="remove", description="將歌曲從播放清單移除")
@app_commands.describe(playlist_name="播放清單名稱", song_title="移除歌曲關鍵字")
@app_commands.autocomplete(playlist_name=playlist_autocompleter,
                           song_title=song_title_autocompleter)
async def remove_from_playlist(interaction: discord.Interaction,
                               playlist_name: str, song_title: str):
    """Removes a song from a specified playlist."""
    guild_id = str(interaction.guild_id)

    if guild_id not in PLAYLISTS or playlist_name not in PLAYLISTS.get(
            guild_id, {}):
        return await interaction.response.send_message(f"找不到播放清單!")

    playlist = PLAYLISTS[guild_id][playlist_name]
    # Find the song to remove by its title
    song_to_remove = next(
        (song for song in playlist if song["title"] == song_title), None)

    if not song_to_remove:
        return await interaction.response.send_message(f"找不到關鍵字!")

    playlist.remove(song_to_remove)  # Remove the song from the list
    save_playlists(PLAYLISTS)  # Save changes to file
    await interaction.response.send_message(
        f"**{song_title}** 已從 **{playlist_name}** 移除")


@bot.tree.command(name="playlist", description="將播放清單歌曲全部加入待播清單")
@app_commands.describe(playlist_name="播放清單名稱", shuffle="是否隨機播放")
@app_commands.autocomplete(playlist_name=playlist_autocompleter)
async def play_custom_playlist(
        interaction: discord.Interaction,
        playlist_name: str,
        shuffle: bool = True):  # Added shuffle parameter with default True
    """Adds all songs from a custom playlist to the current queue."""
    if not interaction.user.voice or not interaction.user.voice.channel:
        return await interaction.response.send_message("你先進語音嘛!")

    guild_id = str(interaction.guild_id)
    if guild_id not in PLAYLISTS or playlist_name not in PLAYLISTS.get(
            guild_id, {}):
        return await interaction.response.send_message(f"找不到播放清單!")

    await interaction.response.defer()  # Defer response

    voice_channel = interaction.user.voice.channel
    voice_client = interaction.guild.voice_client
    if not voice_client:
        voice_client = await voice_channel.connect(
        )  # Connect if not connected
    elif voice_client.channel != voice_channel:
        await voice_client.move_to(voice_channel)  # Move to user's channel

    if guild_id not in SONG_QUEUES:
        SONG_QUEUES[guild_id] = deque()  # Initialize queue

    playlist = PLAYLISTS[guild_id][playlist_name]
    if not playlist:
        return await interaction.followup.send(f"此播放清單是空的!")

    # Apply shuffle if true
    if shuffle:
        random.shuffle(playlist)  # Shuffle the playlist if shuffle is True

    added_count = 0
    for song in playlist:
        # Search for each song in the playlist and add to queue
        song_info, error = await search_song(song["query"])
        if not error:
            audio_url, title, headers = song_info
            SONG_QUEUES[guild_id].append((audio_url, title, headers))
            added_count += 1

    if added_count == 0:
        return await interaction.followup.send(f"壞掉了!")

    await interaction.followup.send(
        f"已從 **{playlist_name}** 加入 **{added_count}** 首歌" +
        (f" (已隨機播放)" if shuffle else " (未隨機播放)")
    )  # Added message about shuffle status

    if not voice_client.is_playing() and not voice_client.is_paused():
        await play_next_song(voice_client, guild_id, interaction.channel)


@bot.tree.command(name="playnext", description="插播歌曲")
@app_commands.describe(song_query="搜尋關鍵字")
async def playnext(interaction: discord.Interaction, song_query: str):
    """Inserts a song at the beginning of the queue and plays it next."""
    if not interaction.user.voice or not interaction.user.voice.channel:
        return await send_or_replace(interaction, "你先進語音嘛！")

    # Check for banned keywords before proceeding with the search
    if await check_banned_keywords(interaction, song_query):
        return  # Exit if a banned keyword is found

    await interaction.response.defer()  # Defer response
    voice_channel = interaction.user.voice.channel
    voice_client = interaction.guild.voice_client
    if voice_client is None:
        voice_client = await voice_channel.connect(
        )  # Connect to voice channel
    elif voice_channel != voice_client.channel:
        await voice_client.move_to(voice_channel
                                   )  # Move to user's voice channel

    song_info, error_message = await search_song(song_query)
    if error_message:
        await interaction.followup.send(error_message)
        return

    audio_url, title, headers = song_info
    guild_id = str(interaction.guild_id)
    if SONG_QUEUES.get(guild_id) is None:
        SONG_QUEUES[guild_id] = deque()  # Initialize queue if not exists

    SONG_QUEUES[guild_id].appendleft(
        (audio_url, title, headers))  # Add song to the front of the queue with headers

    if voice_client.is_playing() or voice_client.is_paused():
        await interaction.followup.send(f"已插播歌曲： **{title}**")
        voice_client.stop()  # Stop current song to play the inserted one next
    else:
        await interaction.followup.send(f"正在播放： **{title}**")
        await play_next_song(voice_client, guild_id, interaction.channel)

    """Bans a specific keyword from being searched."""
    guild_id = str(interaction.guild_id)
    keyword_lower = keyword.lower(
    )  # Convert keyword to lowercase for consistency
    if guild_id not in BANNED_KEYWORDS:
        BANNED_KEYWORDS[guild_id] = set(
        )  # Initialize set for guild if it doesn't exist

async def send_or_replace(interaction: discord.Interaction, content: str):
    """
    Sends a message, replacing the bot's last message in the channel if possible,
    or sends a new one if the interaction response has already been sent.
    """
    key = (interaction.guild.id, interaction.channel.id)
    if key in LAST_MESSAGES:
        try:
            await LAST_MESSAGES[key].delete(
            )  # Attempt to delete the last bot message
        except discord.NotFound:
            pass  # Message already deleted

    if interaction.response.is_done():
        msg = await interaction.followup.send(
            content, ephemeral=False)  # Send as followup if response done
    else:
        await interaction.response.send_message(content, ephemeral=False
                                                )  # Send initial response
        msg = await interaction.original_response(
        )  # Get the original response message
    LAST_MESSAGES[key] = msg  # Store the new message for future replacement
